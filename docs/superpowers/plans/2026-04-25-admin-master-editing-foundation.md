# Admin Master Editing Foundation — Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make every admin master record (items, components, suppliers, supplier-items) fully editable from its own portal page — with safe inline edits, warn-before-save drawers for risky fields, locked dangerous fields, dependency visibility, and archive/restore.

**Architecture:** Eight tranches land progressively on Path B (`/admin/masters/*`), which is already the navigation target. Path A (`/admin/components/[id]`, `/admin/suppliers/[id]`) is redirected in T1 to eliminate the dual-tree. A new `<MasterSummaryCard>` shared component anchors the information architecture on every detail page. All mutations reuse existing `patchEntity()` + `postStatus()` helpers from `src/lib/admin/mutations.ts`.

**Tech Stack:** Next.js 15 App Router, TypeScript, TanStack Query v5, `patchEntity()` / `postStatus()` mutation helpers, `InlineEditCell` for Class S edits (props: `value`, `onSave`, `type`, `format`, `disabled`), new `ClassWEditDrawer` for Class W, Lucide icons, existing `SectionCard` / `DetailPage` / `WorkflowHeader` layout primitives.

**Spec:** `docs/superpowers/specs/2026-04-25-admin-master-editing-foundation-design.md`

**Key confirmed variable names (verified against real files):**
- Component detail (`/admin/masters/components/[id]`): item row = `component`, supplier-items list = `allSi`, query key = `["admin", "masters", "component", component_id]`, existing mutations = `fieldMutation` (supplier-item field PATCH), `promotePrimaryMutation`
- Supplier detail (`/admin/masters/suppliers/[id]`): row = `supplier`, supplier-items list = `allSi`, query key = `["admin", "masters", "supplier", supplier_id]`
- Item detail (`/admin/masters/items/[id]`): row = `row`, supplier-items query = `itemSupplierItemsQuery`, BOM head = `bomHeadQuery`, pack head = `bomHead`, query key = `["admin", "masters", "item", item_id]`

```typescript
// src/components/admin/MasterSummaryCard.tsx
"use client";

import { type ReactNode } from "react";
import { Badge } from "@/components/badges/StatusBadge";

export interface CompletenessItem {
  label: string;
  status: "ok" | "warn" | "error" | "na";
  detail?: string;
}

interface MasterSummaryCardProps {
  name: string;
  code: string;
  entityType: string;
  status: string;
  completeness: CompletenessItem[];
  actions?: ReactNode;
}

function statusTone(status: string): "success" | "warning" | "neutral" | "danger" {
  if (status === "ACTIVE") return "success";
  if (status === "PENDING") return "warning";
  return "neutral";
}

function statusLabel(status: string): string {
  if (status === "ACTIVE") return "Active";
  if (status === "PENDING") return "Pending review";
  if (status === "INACTIVE") return "Archived";
  return status;
}

function CompletenessIcon({ status }: { status: CompletenessItem["status"] }) {
  if (status === "ok") return <span className="text-success-fg">✅</span>;
  if (status === "warn") return <span className="text-warning-fg">⚠️</span>;
  if (status === "error") return <span className="text-danger-fg">❌</span>;
  return <span className="text-fg-subtle">—</span>;
}

export function MasterSummaryCard({
  name,
  code,
  entityType,
  status,
  completeness,
  actions,
}: MasterSummaryCardProps) {
  return (
    <div className="rounded-lg border border-border bg-bg-card p-4 space-y-3">
      <div className="flex flex-wrap items-start justify-between gap-2">
        <div className="min-w-0">
          <div className="flex flex-wrap items-center gap-2 mb-0.5">
            <Badge tone={statusTone(status)} dotted>
              {statusLabel(status)}
            </Badge>
            <span className="text-xs text-fg-subtle">{entityType}</span>
          </div>
          <h2 className="text-lg font-semibold text-fg-strong leading-tight">{name}</h2>
          <p className="text-xs font-mono text-fg-muted mt-0.5">{code}</p>
        </div>
        {actions ? (
          <div className="flex flex-wrap items-center gap-2 shrink-0">{actions}</div>
        ) : null}
      </div>

      {completeness.length > 0 ? (
        <div className="border-t border-border/60 pt-3 space-y-1">
          {completeness.map((item) => (
            <div key={item.label} className="flex items-start gap-2 text-sm">
              <CompletenessIcon status={item.status} />
              <span className="text-fg-muted">
                {item.label}
                {item.detail ? (
                  <span className="text-fg-subtle"> — {item.detail}</span>
                ) : null}
              </span>
            </div>
          ))}
        </div>
      ) : null}
    </div>
  );
}
```

```typescript
// src/components/admin/UsedInRecipes.tsx
"use client";

import { useMemo } from "react";
import Link from "next/link";
import { useQuery, useQueries } from "@tanstack/react-query";
import { Badge } from "@/components/badges/StatusBadge";

interface BomHeadRow {
  bom_head_id: string;
  parent_ref_id: string;
  parent_ref_type: string;
  active_version_id: string | null;
  status: string;
}

interface BomLineRow {
  line_id: string;
  bom_version_id: string;
  final_component_id: string;
  qty_per_l_output: string;
  component_uom: string | null;
}

const MAX_BOM_COUNT_FOR_CLIENT_FETCH = 50;

async function fetchJson<T>(url: string): Promise<T> {
  const res = await fetch(url, { headers: { Accept: "application/json" } });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return (await res.json()) as T;
}

export function UsedInRecipes({ componentId }: { componentId: string }) {
  const headsQuery = useQuery<{ rows: BomHeadRow[]; count: number }>({
    queryKey: ["admin", "bom-heads", "all"],
    queryFn: () => fetchJson("/api/boms/heads?limit=1000"),
    staleTime: 60_000,
  });

  const heads = headsQuery.data?.rows ?? [];
  const activeHeads = heads.filter((h) => h.active_version_id);
  const tooMany = activeHeads.length > MAX_BOM_COUNT_FOR_CLIENT_FETCH;

  const lineQueries = useQueries({
    queries: tooMany
      ? []
      : activeHeads.map((h) => ({
          queryKey: ["admin", "bom-lines", h.active_version_id],
          queryFn: () =>
            fetchJson<{ rows: BomLineRow[] }>(
              `/api/boms/lines?bom_version_id=${encodeURIComponent(h.active_version_id!)}&limit=500`,
            ),
          staleTime: 60_000,
        })),
  });

  const usedInRows = useMemo(() => {
    if (tooMany || lineQueries.some((q) => q.isLoading)) return null;
    const result: Array<{
      bom_head_id: string;
      parent_ref_id: string;
      version_ref: string;
      qty_per_l_output: string;
      component_uom: string | null;
    }> = [];

    activeHeads.forEach((head, i) => {
      const lq = lineQueries[i];
      if (!lq?.data) return;
      lq.data.rows
        .filter((line) => line.final_component_id === componentId)
        .forEach((line) => {
          result.push({
            bom_head_id: head.bom_head_id,
            parent_ref_id: head.parent_ref_id,
            version_ref: head.active_version_id
              ? head.active_version_id.slice(0, 8)
              : "—",
            qty_per_l_output: line.qty_per_l_output,
            component_uom: line.component_uom,
          });
        });
    });
    return result;
  }, [activeHeads, lineQueries, componentId, tooMany]);

  const failedCount = lineQueries.filter((q) => q.isError).length;
  const isLoading = headsQuery.isLoading || lineQueries.some((q) => q.isLoading);

  if (headsQuery.isError) {
    return (
      <div className="p-4 text-sm text-danger-fg">
        Could not load recipe data.{" "}
        <button onClick={() => headsQuery.refetch()} className="underline">
          Retry
        </button>
      </div>
    );
  }

  if (tooMany) {
    return (
      <div className="p-4 rounded-md border border-warning/40 bg-warning-softer text-sm text-warning-fg">
        ⚠️ Recipe lookup is not available in this view — the catalog has more than{" "}
        {MAX_BOM_COUNT_FOR_CLIENT_FETCH} active recipes, which exceeds the client-side limit. A
        backend lookup endpoint is in progress. In the meantime, search for this component in the
        BOM list.
      </div>
    );
  }

  if (isLoading) {
    return <div className="p-4 text-sm text-fg-muted">Loading recipes…</div>;
  }

  if (!usedInRows) return null;

  return (
    <div className="space-y-3">
      {failedCount > 0 ? (
        <div className="rounded-md border border-warning/40 bg-warning-softer p-3 text-sm text-warning-fg">
          {failedCount} recipe section{failedCount !== 1 ? "s" : ""} failed to load.
        </div>
      ) : null}

      {usedInRows.length === 0 ? (
        <div className="p-4 text-sm text-fg-muted">
          This raw material is not currently used in any active recipe.
        </div>
      ) : (
        <>
          <div className="rounded-md border border-warning/40 bg-warning-softer p-3 text-sm text-warning-fg">
            ⚠️ This raw material is used in{" "}
            <strong>
              {usedInRows.length} active recipe{usedInRows.length !== 1 ? "s" : ""}
            </strong>
            . Changes affect production planning. Open a recipe before modifying.
          </div>
          <p className="px-1 text-xs text-fg-subtle">
            Version ref shows first 8 chars of version ID. Full labels available when the backend
            lookup endpoint is live.
          </p>
          <div className="overflow-x-auto">
            <table className="w-full border-collapse text-sm">
              <thead>
                <tr className="border-b border-border/70 bg-bg-subtle/60">
                  <th className="px-3 py-2 text-left text-3xs font-semibold uppercase tracking-sops text-fg-subtle">
                    Recipe
                  </th>
                  <th className="px-3 py-2 text-left text-3xs font-semibold uppercase tracking-sops text-fg-subtle">
                    Version ref
                  </th>
                  <th className="px-3 py-2 text-left text-3xs font-semibold uppercase tracking-sops text-fg-subtle">
                    Qty / L output
                  </th>
                  <th className="px-3 py-2 text-left text-3xs font-semibold uppercase tracking-sops text-fg-subtle">
                    Unit
                  </th>
                  <th className="px-3 py-2 text-right text-3xs font-semibold uppercase tracking-sops text-fg-subtle" />
                </tr>
              </thead>
              <tbody>
                {usedInRows.map((row, i) => (
                  <tr
                    key={i}
                    className="border-b border-border/40 last:border-b-0 hover:bg-bg-subtle/40"
                  >
                    <td className="px-3 py-2 font-mono text-xs text-fg">{row.parent_ref_id}</td>
                    <td className="px-3 py-2 text-xs font-mono text-fg-muted">{row.version_ref}</td>

          </div>
        </>
      )}
    </div>
  );
}
```

```markdown
# Contract Gap #1: GET /api/components/:component_id/used-in-recipes

**Status:** Needed by portal T3. Portal ships client-side fallback until this lands.
**Owner:** Backend (W1)
**Priority:** High — client-side fallback fails for catalogs with > 50 active BOMs, and displays UUID version refs instead of human labels.

## Endpoint

GET /api/v1/queries/components/:component_id/used-in-recipes?mode=ui|gate

## Auth

admin / planner / viewer (read-only)

## Query params

- `mode=ui` (default) — returns ACTIVE + DRAFT versions
- `mode=gate` — returns ACTIVE only (for archive precheck); omits draft_count

## Response 200

{
  "rows": [
    {
      "bom_head_id": "BOM-BASE-AME-REG",
      "parent_ref_id": "BASE-AME-REG",
      "parent_ref_type": "component",
      "parent_name": "Base Americano Regular",
      "bom_version_id": "uuid",
      "version_label": "V4_COST_FILE",
      "version_status": "ACTIVE",
      "line_id": "uuid",
      "qty_per_l_output": "0.0042",
      "component_uom": "UNIT"
    }
  ],
  "count": 14,
  "active_count": 12,
  "draft_count": 2
}

## SQL (UI mode)

SELECT bh.bom_head_id, bh.parent_ref_id, bh.parent_ref_type,
       COALESCE(i.item_name, c.component_name) AS parent_name,
       bv.bom_version_id, bv.version_label, bv.status AS version_status,
       bl.line_id, bl.qty_per_l_output, bl.component_uom
FROM private_core.bom_lines bl
JOIN private_core.bom_version bv ON bv.bom_version_id = bl.bom_version_id
JOIN private_core.bom_head  bh ON bh.bom_head_id = bv.bom_head_id
LEFT JOIN private_core.items i
  ON bh.parent_ref_type = 'item' AND i.item_id = bh.parent_ref_id
LEFT JOIN private_core.components c
  ON bh.parent_ref_type = 'component' AND c.component_id = bh.parent_ref_id
WHERE bl.final_component_id = :component_id
  AND bv.status IN ('ACTIVE','DRAFT')
ORDER BY bv.status, bh.parent_ref_id;

For gate mode: change IN ('ACTIVE','DRAFT') to = 'ACTIVE'.
```

```typescript
"use client";

import { useState } from "react";
import Link from "next/link";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { WorkflowHeader } from "@/components/workflow/WorkflowHeader";
import { SectionCard } from "@/components/workflow/SectionCard";
import { Badge } from "@/components/badges/StatusBadge";
import { ClassWEditDrawer } from "@/components/admin/ClassWEditDrawer";
import { AdminMutationError, postStatus } from "@/lib/admin/mutations";
import { useSession } from "@/lib/auth/session-provider";

type EntityType = "components" | "items" | "suppliers" | "supplier-items";

async function fetchJson<T>(url: string): Promise<T> {
  const res = await fetch(url, { headers: { Accept: "application/json" } });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return (await res.json()) as T;
}

function idFromRow(type: EntityType, row: Record<string, unknown>): string {
  if (type === "components") return String(row.component_id ?? "");
  if (type === "items") return String(row.item_id ?? "");
  if (type === "suppliers") return String(row.supplier_id ?? "");
  return String(row.supplier_item_id ?? "");
}

function nameFromRow(type: EntityType, row: Record<string, unknown>): string {
  if (type === "components") return String(row.component_name ?? "");
  if (type === "items") return String(row.item_name ?? "");
  if (type === "suppliers") return String(row.supplier_name_official ?? "");
  return `${row.supplier_id} → ${row.component_id ?? row.item_id ?? "?"}`;
}

function detailHref(type: EntityType, id: string): string | null {
  if (type === "components") return `/admin/masters/components/${encodeURIComponent(id)}`;
  if (type === "items") return `/admin/masters/items/${encodeURIComponent(id)}`;
  if (type === "suppliers") return `/admin/masters/suppliers/${encodeURIComponent(id)}`;
  return null;
}

function statusUrl(type: EntityType, id: string): string {
  if (type === "supplier-items") return `/api/supplier-items/${encodeURIComponent(id)}/status`;
  if (type === "components") return `/api/components/${encodeURIComponent(id)}/status`;
  if (type === "items") return `/api/items/${encodeURIComponent(id)}/status`;
  return `/api/suppliers/${encodeURIComponent(id)}/status`;
}

const ENTITY_QUERY_URLS: Record<EntityType, string> = {
  components: "/api/components?status=INACTIVE&limit=500",
  items: "/api/items?status=INACTIVE&limit=500",
  suppliers: "/api/suppliers?status=INACTIVE&limit=500",
  "supplier-items": "/api/supplier-items?status=INACTIVE&limit=500",
};

  { key: "components", label: "Raw materials" },
  { key: "items", label: "Products" },
  { key: "suppliers", label: "Suppliers" },
  { key: "supplier-items", label: "Sourcing links" },
];

function ArchiveTab({ type, isAdmin }: { type: EntityType; isAdmin: boolean }) {
  const queryClient = useQueryClient();
  const [restoringRow, setRestoringRow] = useState<{
    id: string;
    updated_at: string;
  } | null>(null);
  const [banner, setBanner] = useState<string | null>(null);

  const q = useQuery<{ rows: Record<string, unknown>[]; count: number }>({
    queryKey: ["admin", "archive", type],
    queryFn: () => fetchJson(ENTITY_QUERY_URLS[type]),
  });

  const restoreMutation = useMutation({
    mutationFn: (args: { id: string; updated_at: string }) =>
      postStatus({
        url: statusUrl(type, args.id),
        status: "ACTIVE",
        ifMatchUpdatedAt: args.updated_at,
      }),
    onSuccess: (_data, vars) => {
      setRestoringRow(null);
      setBanner(`Restored ${vars.id}.`);
      void queryClient.invalidateQueries({ queryKey: ["admin", "archive", type] });
      void queryClient.invalidateQueries({ queryKey: ["admin", type] });
    },
    onError: (err: Error) => {
      const msg =
        err instanceof AdminMutationError ? err.message : err.message;
      setBanner(`Error: ${msg}`);
    },
  });

  if (q.isLoading) return <div className="p-5 text-sm text-fg-muted">Loading…</div>;
  if (q.isError)
    return (
      <div className="p-5 text-sm text-danger-fg">Could not load archived records.</div>
    );

  const rows = q.data?.rows ?? [];

  return (
    <div>
      {banner ? (
        <div className="m-4 rounded-md border border-success/40 bg-success-softer p-3 text-sm text-success-fg">
          {banner}
        </div>
      ) : null}

      <ClassWEditDrawer
        open={restoringRow != null}
        onClose={() => setRestoringRow(null)}
        title="Restore this record?"
        warning="Restoring sets this record back to Active. It will reappear in active lists and may be included in the next planning run."
        onSave={async () => {
          if (!restoringRow) return;
          await restoreMutation.mutateAsync(restoringRow);
        }}
        isSaving={restoreMutation.isPending}
        error={restoreMutation.isError ? (restoreMutation.error as Error).message : null}
      >
        <p className="text-sm text-fg-muted">
          Record: <strong>{restoringRow?.id}</strong>
        </p>
      </ClassWEditDrawer>

      {rows.length === 0 ? (
        <div className="p-5 text-sm text-fg-muted">
          No archived records in this category.
        </div>
      ) : (
        <div className="overflow-x-auto">
          <table className="w-full border-collapse text-sm">
            <thead>
              <tr className="border-b border-border/70 bg-bg-subtle/60">
                <th className="px-3 py-2 text-left text-3xs font-semibold uppercase tracking-sops text-fg-subtle">
                  Name
                </th>
                <th className="px-3 py-2 text-left text-3xs font-semibold uppercase tracking-sops text-fg-subtle">
                  Code
                </th>
                <th className="px-3 py-2 text-left text-3xs font-semibold uppercase tracking-sops text-fg-subtle">
                  Archived
                </th>
                <th className="px-3 py-2 text-right text-3xs font-semibold uppercase tracking-sops text-fg-subtle">
                  Actions
                </th>
              </tr>
            </thead>
            <tbody>
              {rows.map((rawRow) => {
                const id = idFromRow(type, rawRow);
                const name = nameFromRow(type, rawRow);
                const updated_at = String(rawRow.updated_at ?? "");
                const href = detailHref(type, id);
                return (
                  <tr
                    key={id}
                    className="border-b border-border/40 last:border-b-0 hover:bg-bg-subtle/40"
                  >
                    <td className="px-3 py-2 text-fg-strong">

                    </td>

                          type="button"

                  </tr>

          </table>
        </div>
      )}

  );
```
